```php
<?php

namespace App\Http\Livewire\Admin;

use App\Models\Post;
use App\Models\Comment;
use App\Models\View;
use App\Models\Reaction;
use App\Models\Share;
use Carbon\Carbon;
use Livewire\Component;
use Illuminate\Support\Facades\Cache;
use Illuminate\Support\Facades\DB;

class BlogAnalytics extends Component
{
    public $dateFrom;
    public $dateTo;
    public $selectedPost = null;
    public $selectedAuthor = null;
    public $statusFilter = 'all';
    
    public $stats = [];
    public $chartData = [];
    public $topPosts = [];
    public $reactionBreakdown = [];
    public $sharesPlatform = [];

    protected $queryString = ['dateFrom', 'dateTo', 'selectedPost', 'selectedAuthor', 'statusFilter'];

    public function mount()
    {
        $this->dateFrom = Carbon::now()->subDays(30)->format('Y-m-d');
        $this->dateTo = Carbon::now()->format('Y-m-d');
        $this->loadStats();
    }

    public function render()
    {
        return view('livewire.admin.blog-analytics', [
            'posts' => Post::select('id', 'title')->get(),
            'authors' => \App\Models\User::whereHas('posts')->select('id', 'name')->get()
        ])->layout('layouts.admin');
    }

    public function loadStats()
    {
        $cacheKey = "blog_analytics_{$this->dateFrom}_{$this->dateTo}_{$this->selectedPost}_{$this->selectedAuthor}_{$this->statusFilter}";
        
        $data = Cache::remember($cacheKey, 60, function () {
            $from = Carbon::parse($this->dateFrom)->startOfDay();
            $to = Carbon::parse($this->dateTo)->endOfDay();

            // Base queries
            $postsQuery = Post::whereBetween('created_at', [$from, $to]);
            $commentsQuery = Comment::whereBetween('created_at', [$from, $to]);
            $viewsQuery = View::whereBetween('created_at', [$from, $to]);
            $reactionsQuery = Reaction::whereBetween('created_at', [$from, $to]);
            $sharesQuery = Share::whereBetween('created_at', [$from, $to]);

            // Apply filters
            if ($this->selectedPost) {
                $commentsQuery->where('post_id', $this->selectedPost);
                $viewsQuery->where('viewable_id', $this->selectedPost)->where('viewable_type', Post::class);
                $reactionsQuery->where('reactable_id', $this->selectedPost)->where('reactable_type', Post::class);
                $sharesQuery->where('shareable_id', $this->selectedPost)->where('shareable_type', Post::class);
            }

            if ($this->selectedAuthor) {
                $postIds = Post::where('user_id', $this->selectedAuthor)->pluck('id');
                $commentsQuery->whereIn('post_id', $postIds);
            }

            if ($this->statusFilter !== 'all') {
                $commentsQuery->where('status', $this->statusFilter);
            }

            // Aggregate stats
            $stats = [
                'total_posts' => $postsQuery->count(),
                'total_comments' => $commentsQuery->count(),
                'pending_comments' => Comment::whereBetween('created_at', [$from, $to])->pending()->count(),
                'total_views' => $viewsQuery->count(),
                'total_reactions' => $reactionsQuery->count(),
                'total_shares' => $sharesQuery->count(),
            ];

            // Daily views for chart
            $viewsDaily = View::selectRaw('DATE(created_at) as date, COUNT(*) as count')
                ->whereBetween('created_at', [$from, $to])
                ->where('viewable_type', Post::class)
                ->groupBy('date')
                ->orderBy('date')
                ->get();

            $chartData = [
                'dates' => $viewsDaily->pluck('date')->map(fn($d) => Carbon::parse($d)->format('d/m'))->toArray(),
                'views' => $viewsDaily->pluck('count')->toArray(),
            ];

            // Top 5 posts
            $topPosts = Post::withCount(['views', 'comments', 'reactions'])
                ->whereBetween('created_at', [$from, $to])
                ->orderBy('views_count', 'desc')
                ->take(5)
                ->get()
                ->map(function ($post) {
                    return [
                        'title' => $post->title,
                        'views' => $post->views_count,
                        'comments' => $post->comments_count,
                        'reactions' => $post->reactions_count,
                    ];
                });

            // Reactions breakdown
            $reactionBreakdown = Reaction::selectRaw('type, COUNT(*) as count')
                ->whereBetween('created_at', [$from, $to])
                ->where('reactable_type', Post::class)
                ->groupBy('type')
                ->get()
                ->mapWithKeys(fn($r) => [$r->type => $r->count]);

            // Shares by platform
            $sharesPlatform = Share::selectRaw('platform, COUNT(*) as count')
                ->whereBetween('created_at', [$from, $to])
                ->groupBy('platform')
                ->get()
                ->mapWithKeys(fn($s) => [$s->platform => $s->count]);

            return [
                'stats' => $stats,
                'chartData' => $chartData,
                'topPosts' => $topPosts,
                'reactionBreakdown' => $reactionBreakdown,
                'sharesPlatform' => $sharesPlatform,
            ];
        });

        $this->stats = $data['stats'];
        $this->chartData = $data['chartData'];
        $this->topPosts = $data['topPosts'];
        $this->reactionBreakdown = $data['reactionBreakdown'];
        $this->sharesPlatform = $data['sharesPlatform'];
    }

    public function refreshStats()
    {
        Cache::forget("blog_analytics_{$this->dateFrom}_{$this->dateTo}_{$this->selectedPost}_{$this->selectedAuthor}_{$this->statusFilter}");
        $this->loadStats();
        $this->dispatchBrowserEvent('stats-refreshed');
    }

    public function exportCsv()
    {
        return redirect()->route('admin.analytics.export', [
            'from' => $this->dateFrom,
            'to' => $this->dateTo,
            'post' => $this->selectedPost,
            'author' => $this->selectedAuthor,
            'status' => $this->statusFilter,
        ]);
    }

    public function updatedDateFrom()
    {
        $this->loadStats();
    }

    public function updatedDateTo()
    {
        $this->loadStats();
    }

    public function updatedSelectedPost()
    {
        $this->loadStats();
    }

    public function updatedSelectedAuthor()
    {
        $this->loadStats();
    }

    public function updatedStatusFilter()
    {
        $this->loadStats();
    }
}
```

```php
<?php

namespace App\Http\Livewire\Admin;

use App\Models\Comment;
use App\Models\Post;
use Livewire\Component;
use Livewire\WithPagination;

class CommentManagement extends Component
{
    use WithPagination;

    public $search = '';
    public $status = 'all';
    public $post_id = '';
    public $sortField = 'created_at';
    public $sortDirection = 'desc';
    public $perPage = 15;
    public $selected = [];
    public $selectAll = false;

    protected $queryString = ['search', 'status', 'post_id', 'sortField', 'sortDirection'];
    protected $paginationTheme = 'tailwind';

    public function render()
    {
        $comments = Comment::with(['post:id,title', 'user:id,name,email'])
            ->when($this->search, fn($q) => $q->search($this->search))
            ->when($this->status !== 'all', fn($q) => $q->where('status', $this->status))
            ->when($this->post_id, fn($q) => $q->where('post_id', $this->post_id))
            ->orderBy($this->sortField, $this->sortDirection)
            ->paginate($this->perPage);

        return view('livewire.admin.comment-management', [
            'comments' => $comments,
            'posts' => Post::select('id', 'title')->get()
        ])->layout('layouts.admin');
    }

    public function updatingSearch()
    {
        $this->resetPage();
    }

    public function updatingStatus()
    {
        $this->resetPage();
    }

    public function updatingPostId()
    {
        $this->resetPage();
    }

    public function sortBy($field)
    {
        if ($this->sortField === $field) {
            $this->sortDirection = $this->sortDirection === 'asc' ? 'desc' : 'asc';
        } else {
            $this->sortField = $field;
            $this->sortDirection = 'asc';
        }
    }

    public function approve($id)
    {
        $comment = Comment::findOrFail($id);
        $comment->update(['status' => 'approved', 'moderated_at' => now(), 'moderated_by' => auth()->id()]);
        $this->dispatchBrowserEvent('comment-approved', ['message' => 'Commentaire approuvé avec succès']);
    }

    public function reject($id)
    {
        $comment = Comment::findOrFail($id);
        $comment->update(['status' => 'rejected', 'moderated_at' => now(), 'moderated_by' => auth()->id()]);
        $this->dispatchBrowserEvent('comment-rejected', ['message' => 'Commentaire rejeté']);
    }

    public function markSpam($id)
    {
        $comment = Comment::findOrFail($id);
        $comment->update(['status' => 'spam', 'moderated_at' => now(), 'moderated_by' => auth()->id()]);
        $this->dispatchBrowserEvent('comment-spam', ['message' => 'Commentaire marqué comme spam']);
    }

    public function delete($id)
    {
        $this->dispatchBrowserEvent('confirm-delete', ['id' => $id]);
    }

    public function confirmDelete($id)
    {
        Comment::findOrFail($id)->delete();
        $this->dispatchBrowserEvent('comment-deleted', ['message' => 'Commentaire supprimé']);
    }

    public function openDetails($id)
    {
        $this->emit('openCommentDrawer', $id);
    }

    public function bulkDelete()
    {
        if (empty($this->selected)) {
            $this->dispatchBrowserEvent('alert', ['message' => 'Aucun commentaire sélectionné']);
            return;
        }

        $this->dispatchBrowserEvent('confirm-bulk-delete');
    }

    public function confirmBulkDelete()
    {
        Comment::whereIn('id', $this->selected)->delete();
        $this->selected = [];
        $this->selectAll = false;
        $this->dispatchBrowserEvent('comments-deleted', ['message' => count($this->selected) . ' commentaires supprimés']);
    }

    public function updatedSelectAll($value)
    {
        if ($value) {
            $this->selected = Comment::when($this->search, fn($q) => $q->search($this->search))
                ->when($this->status !== 'all', fn($q) => $q->where('status', $this->status))
                ->when($this->post_id, fn($q) => $q->where('post_id', $this->post_id))
                ->pluck('id')
                ->toArray();
        } else {
            $this->selected = [];
        }
    }
}
```

```php
<?php

namespace App\Http\Livewire\Admin;

use App\Models\Comment;
use Livewire\Component;

class CommentDrawer extends Component
{
    public $isOpen = false;
    public $comment = null;
    public $replyContent = '';

    protected $listeners = ['openCommentDrawer' => 'open'];

    public function render()
    {
        return view('livewire.admin.comment-drawer');
    }

    public function open($commentId)
    {
        $this->comment = Comment::with(['post:id,title', 'user:id,name,email,avatar', 'replies.user', 'parent', 'moderator'])
            ->withCount(['likes', 'reports'])
            ->findOrFail($commentId);
        $this->isOpen = true;
    }

    public function close()
    {
        $this->isOpen = false;
        $this->comment = null;
        $this->replyContent = '';
    }

    public function reply()
    {
        $this->validate([
            'replyContent' => 'required|min:3|max:1000'
        ], [
            'replyContent.required' => 'La réponse ne peut pas être vide',
            'replyContent.min' => 'La réponse doit contenir au moins 3 caractères',
            'replyContent.max' => 'La réponse ne peut pas dépasser 1000 caractères'
        ]);

        Comment::create([
            'post_id' => $this->comment->post_id,
            'user_id' => auth()->id(),
            'parent_id' => $this->comment->id,
            'content' => $this->replyContent,
            'status' => 'approved',
            'ip_address' => request()->ip(),
            'user_agent' => request()->userAgent()
        ]);

        $this->replyContent = '';
        $this->open($this->comment->id);
        $this->dispatchBrowserEvent('reply-sent', ['message' => 'Réponse envoyée avec succès']);
    }

    public function approve()
    {
        $this->comment->update(['status' => 'approved', 'moderated_at' => now(), 'moderated_by' => auth()->id()]);
        $this->open($this->comment->id);
        $this->emit('$refresh');
    }

    public function reject()
    {
        $this->comment->update(['status' => 'rejected', 'moderated_at' => now(), 'moderated_by' => auth()->id()]);
        $this->open($this->comment->id);
        $this->emit('$refresh');
    }

    public function markSpam()
    {
        $this->comment->update(['status' => 'spam', 'moderated_at' => now(), 'moderated_by' => auth()->id()]);
        $this->open($this->comment->id);
        $this->emit('$refresh');
    }

    public function delete()
    {
        $this->comment->delete();
        $this->close();
        $this->emit('$refresh');
        $this->dispatchBrowserEvent('comment-deleted-drawer', ['message' => 'Commentaire supprimé']);
    }
}
```

```php
<?php

namespace App\Models;

use Illuminate\Database\Eloquent\Factories\HasFactory;
use Illuminate\Database\Eloquent\Model;
use Illuminate\Database\Eloquent\SoftDeletes;

class Comment extends Model
{
    use HasFactory, SoftDeletes;

    protected $fillable = [
        'post_id',
        'user_id',
        'parent_id',
        'content',
        'author_name',
        'author_email',
        'status',
        'ip_address',
        'user_agent',
        'moderated_at',
        'moderated_by'
    ];

    protected $casts = [
        'moderated_at' => 'datetime',
        'created_at' => 'datetime',
        'updated_at' => 'datetime'
    ];

    protected $with = [];

    // Relationships
    public function post()
    {
        return $this->belongsTo(Post::class);
    }

    public function user()
    {
        return $this->belongsTo(User::class);
    }

    public function parent()
    {
        return $this->belongsTo(Comment::class, 'parent_id');
    }

    public function replies()
    {
        return $this->hasMany(Comment::class, 'parent_id')->latest();
    }

    public function moderator()
    {
        return $this->belongsTo(User::class, 'moderated_by');
    }

    public function likes()
    {
        return $this->morphMany(Reaction::class, 'reactable')->where('type', 'like');
    }

    public function reports()
    {
        return $this->hasMany(CommentReport::class);
    }

    // Scopes
    public function scopePending($query)
    {
        return $query->where('status', 'pending');
    }

    public function scopeApproved($query)
    {
        return $query->where('status', 'approved');
    }

    public function scopeRejected($query)
    {
        return $query->where('status', 'rejected');
    }

    public function scopeSpam($query)
    {
        return $query->where('status', 'spam');
    }

    public function scopeByPost($query, $postId)
    {
        return $query->where('post_id', $postId);
    }

    public function scopeSearch($query, $term)
    {
        return $query->where(function ($q) use ($term) {
            $q->where('content', 'ILIKE', "%{$term}%")
              ->orWhere('author_name', 'ILIKE', "%{$term}%")
              ->orWhere('author_email', 'ILIKE', "%{$term}%")
              ->orWhereHas('user', function ($userQuery) use ($term) {
                  $userQuery->where('name', 'ILIKE', "%{$term}%")
                            ->orWhere('email', 'ILIKE', "%{$term}%");
              });
        });
    }

    public function scopeWithCounts($query)
    {
        return $query->withCount(['likes', 'reports']);
    }
}
```

```php
namespace Database\Seeders;

use App\Models\Comment;
use App\Models\Post;
use App\Models\User;
use App\Models\Reaction;
use App\Models\CommentReport;
use Illuminate\Database\Seeder;

class CommentSeeder extends Seeder
{
    public function run()
    {
        $posts = Post::all();
        $users = User::all();

        if ($posts->isEmpty() || $users->isEmpty()) {
            $this->command->warn('Please seed Posts and Users first.');
            return;
        }

        // Create 100 comments
        Comment::factory()
            ->count(100)
            ->create([
                'post_id' => fn() => $posts->random()->id,
                'user_id' => fn() => $users->random()->id,
            ])
            ->each(function ($comment) use ($users) {
                // Add likes (reactions)
                $likeCount = rand(0, 20);
                for ($i = 0; $i < $likeCount; $i++) {
                    Reaction::create([
                        'reactable_type' => Comment::class,
                        'reactable_id' => $comment->id,
                        'user_id' => $users->random()->id,
                        'type' => 'like'
                    ]);
                }

                // Add reports for some comments
                if (rand(1, 10) > 7) {
                    $reportCount = rand(1, 5);
                    for ($i = 0; $i < $reportCount; $i++) {
                        CommentReport::create([
                            'comment_id' => $comment->id,
                            'user_id' => $users->random()->id,
                            'reason' => collect(['spam', 'inappropriate', 'offensive', 'other'])->random(),
                            'description' => 'Test report ' . rand(1, 1000)
                        ]);
                    }
                }

                // Add replies to some comments
                if (rand(1, 10) > 6) {
                    Comment::factory()
                        ->count(rand(1, 3))
                        ->create([
                            'post_id' => $comment->post_id,
                            'parent_id' => $comment->id,
                            'user_id' => $users->random()->id,
                            'status' => 'approved'
                        ]);
                }
            });

        $this->command->info('Comments seeded successfully with likes, reports, and replies!');
    }
```

```php
        $data[] = ['Métrique', 'Valeur'];

        $data[] = ['Vues totales', $viewsCount];

        $data[] = [];

        // Top posts
        $data[] = ['Top 10 des Articles'];
        $data[] = ['Titre', 'Vues', 'Commentaires', 'Réactions'];

            ->orderBy('views_count', 'desc')

        }

            }
            
            fclose($file);
```
