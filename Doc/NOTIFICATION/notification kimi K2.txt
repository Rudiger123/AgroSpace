Below is a **complete, paste-ready** notification suite for the module  
**“Offres”** – event **“NewOfferCreated”** – using **database + mail + broadcast**.  
All class / file names are in English; all user-visible strings are in French.

3. LISTENER
--------------------------------------------------
File: `app/Listeners/Offers/SendNewOfferNotification.php`

```php
<?php
declare(strict_types=1);

namespace App\Listeners\Offers;

use App\Events\Offers\NewOfferCreated;
use App\Models\User;
use App\Notifications\Offers\NewOfferNotification;
use Illuminate\Contracts\Queue\ShouldQueue;

class SendNewOfferNotification implements ShouldQueue
{
    public function handle(NewOfferCreated $event): void
    {
        // Destinataires : tous les utilisateurs abonnés à la newsletter
        User::query()
            ->where('newsletter_subscribed', true)
            ->where('id', '!=', $event->creator->id) // optionnel
            ->cursor()
            ->each(fn (User $user) => $user->notify(
                new NewOfferNotification($event->creator, $event->offer)
            ));
    }
}
```

4. NOTIFICATION
--------------------------------------------------
File: `app/Notifications/Offers/NewOfferNotification.php`

```php
<?php
declare(strict_types=1);

namespace App\Notifications\Offers;

use App\Models\Offer;
use App\Models\User;
use Illuminate\Bus\Queueable;
use Illuminate\Contracts\Broadcasting\ShouldBroadcast;
use Illuminate\Notifications\Messages\BroadcastMessage;
use Illuminate\Notifications\Messages\MailMessage;
use Illuminate\Notifications\Notification;

class NewOfferNotification extends Notification implements ShouldBroadcast
{
    use Queueable;

    public function __construct(
        public readonly User  $creator,
        public readonly Offer $offer
    ) {}

    public function via($notifiable): array
    {
        return ['database', 'mail', 'broadcast'];
    }

    /* ---------- DATABASE ---------- */
    public function toDatabase($notifiable): array
    {
        return [
            'offer_id'    => $this->offer->id,
            'offer_title' => $this->offer->title,
            'creator_name'=> $this->creator->name,
            'created_at'  => now()->toIso8601String(),
        ];
    }

    /* ---------- MAIL ---------- */
    public function toMail($notifiable): MailMessage
    {
        return (new MailMessage)
            ->subject('Nouvelle offre disponible')
            ->greeting('Bonjour ' . $notifiable->name . ' !')
            ->line('Une nouvelle offre vient d’être publiée :')
            ->line('**' . $this->offer->title . '**')
            ->action('Voir l’offre', url('/offers/' . $this->offer->id))
            ->line('Merci de votre confiance.');
    }

    /* ---------- BROADCAST ---------- */
    public function toBroadcast($notifiable): BroadcastMessage
    {
        return new BroadcastMessage([
            'id'            => $this->id,
            'offer_id'      => $this->offer->id,
            'offer_title'   => $this->offer->title,
            'creator_name'  => $this->creator->name,
            'created_at'    => now()->toIso8601String(),
            'read_at'       => null,
        ]);
    }
}
```

5. LIVEWIRE COMPONENTS

A. **Bell + counter + dropdown**  
File: `app/Livewire/Notifications/NotificationBell.php`

```php
<?php
declare(strict_types=1);

namespace App\Livewire\Notifications;

use Illuminate\Support\Facades\Auth;
use Livewire\Component;

class NotificationBell extends Component
{
    public $unreadCount = 0;

    protected $listeners = ['echo-private:users.{userId},NotificationSent' => 'refreshCount'];

    public function mount(): void
    {
        $this->refreshCount();
    }

    public function refreshCount(): void
    {
        $this->unreadCount = Auth::user()->unreadNotifications()->count();
    }

    public function render()
    {
        return view('livewire.notifications.notification-bell');
    }
}
```

B. **List + mark-as-read**  
File: `app/Livewire/Notifications/NotificationList.php`

```php
<?php
declare(strict_types=1);

namespace App\Livewire\Notifications;

use Illuminate\Support\Facades\Auth;
use Livewire\Component;

class NotificationList extends Component
{
    public $notifications = [];

    protected $listeners = ['echo-private:users.{userId},NotificationSent' => 'load'];

    public function mount(): void
    {
        $this->load();
    }

    public function load(): void
    {
        $this->notifications = Auth::user()
            ->notifications()
            ->latest()
            ->take(15)
            ->get()
            ->toArray();
    }

    public function markAsRead(string $id): void
    {
        Auth::user()->notifications()->where('id', $id)->first()?->markAsRead();
        $this->load();
        $this->dispatch('refresh-bell');
    }

    public function markAllAsRead(): void
    {
        Auth::user()->unreadNotifications->markAsRead();
        $this->load();
        $this->dispatch('refresh-bell');
    }

    public function render()
    {
        return view('livewire.notifications.notification-list');
    }
}
```

C. **Broadcast configuration** (Laravel 12 + Laravel Echo + Pusher/reverb)

.env example

```
BROADCAST_DRIVER=reverb
REVERB_APP_ID=local
REVERB_APP_KEY=local
REVERB_APP_SECRET=local
REVERB_HOST=127.0.0.1
REVERB_PORT=8080
```

Install & start Reverb

```bash
php artisan reverb:install
php artisan reverb:start
```

D. **Echo inclusion** (Bootstrap JS)

```js
import Echo from 'laravel-echo';
import Pusher from 'pusher-js';
window.Pusher = Pusher;

window.Echo = new Echo({
    broadcaster: 'reverb',
    key: import.meta.env.VITE_REVERB_APP_KEY,
    wsHost: import.meta.env.VITE_REVERB_HOST,
    wsPort: import.meta.env.VITE_REVERB_PORT,
    forceTLS: false,
    disableStats: true,
});
```

```php
use App\Events\Offers\NewOfferCreated;
use App\Models\Offer;

public function store(Request $request)
{
    $offer = Offer::create($request->validated());
    
    event(new NewOfferCreated(auth()->user(), $offer));

    return redirect()->route('offers.index')->with('success', 'Offre publiée.');
}
```

9. RAPPEL DES CANAUX / CONTENU FRANÇAIS
--------------------------------------------------
- **Database** : titre + auteur + date  
- **Mail** : sujet « Nouvelle offre disponible » + bouton « Voir l’offre »  
- **Broadcast** : même payload que DB → mise-à-jour temps-réel du compteur & liste

Tout est copier-coller dans un projet TALL v12 ; exécutez :

```bash
php artisan config:clear
php artisan route:clear
php artisan queue:work          # si queue utilisée
php artisan reverb:start        # ou pusher
```

et vos utilisateurs abonnés recevront la notification **NewOfferCreated** sur les trois canaux, avec interface entièrement francophone.
